//! Parse a deliberately small semicolon-separated format, not general CSV.
use std::error::Error;
use std::fmt;

#[derive(Debug, PartialEq)]
struct Reading {
    station: String,
    temperature: f64,
}

#[derive(Debug, PartialEq)]
struct ParseError {
    line: usize,
    message: &'static str,
}

impl fmt::Display for ParseError {
    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(formatter, "line {}: {}", self.line, self.message)
    }
}

impl Error for ParseError {}

fn parse_readings(input: &str) -> Result<Vec<Reading>, ParseError> {
    let mut readings = Vec::new();
    for (index, raw) in input.lines().enumerate() {
        let line = raw.trim();
        if line.is_empty() || line.starts_with('#') {
            continue;
        }
        let failure = |message| ParseError { line: index + 1, message };
        let (name, number) = line
            .split_once(';')
            .ok_or_else(|| failure("expected station;temperature"))?;
        if name.trim().is_empty() {
            return Err(failure("station name must not be empty"));
        }
        // No quoted fields or embedded separators are accepted by this format.
        if name.contains('"') || number.contains([';', '"']) {
            return Err(failure("quoted or additional fields are unsupported"));
        }
        let temperature: f64 = number
            .trim()
            .parse()
            .map_err(|_| failure("temperature is not a number"))?;
        if !temperature.is_finite() {
            return Err(failure("temperature must be finite"));
        }
        readings.push(Reading { station: name.trim().to_owned(), temperature });
    }
    Ok(readings)
}

fn main() -> Result<(), Box<dyn Error>> {
    let input = "# Mesures du matin\nNorth;12.5\nSouth;15.0\n";
    for reading in parse_readings(input)? {
        println!("{}: {:.1} °C", reading.station, reading.temperature);
    }
    Ok(())
}

#[cfg(test)]
mod tests {
    use super::parse_readings;

    #[test]
    fn preserves_line_numbers_in_errors() {
        let error = parse_readings("# comment\nValid;3.5\nInvalid\n").unwrap_err();
        assert_eq!(error.line, 3);
    }

    #[test]
    fn accepts_whitespace_and_rejects_infinity() {
        let data = parse_readings("\n West ; -2.5 \n").unwrap();
        assert_eq!(data[0].station, "West");
        assert_eq!(data[0].temperature, -2.5);
        assert!(parse_readings("West;inf").is_err());
    }
}
